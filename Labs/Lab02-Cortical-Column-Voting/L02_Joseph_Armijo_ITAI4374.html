def create_sensory_input(true_object, num_columns, noise_level=0.3):
    """
    Creates evidence matrix for all columns.
    Parameters:
    - true_object: int, index of the actual object (0 to 4)
    - num_columns: int, number of columns to simulate
    - noise_level: float, amount of random noise (0 to 1)
    Returns:
    - evidence: numpy array of shape (num_columns, NUM_OBJECTS)
    """
    # TODO 1: Create matrix filled with random noise
    evidence = np.random.uniform(0, noise_level, size=(num_columns, NUM_OBJECTS))

    # TODO 2: Add signal for the true object
    for col in range(num_columns):
        signal = np.random.uniform(0.5, 1.0)
        evidence[col, true_object] += signal

    # TODO 3: Normalize each row to sum to 1
    row_sums = evidence.sum(axis=1, keepdims=True)
    evidence = evidence / row_sums

    return evidence

Voting Round

In [10]:
def voting_round(beliefs, evidence, vote_weight=0.3):
    """
    One round of voting among columns.
    Parameters:
    - beliefs: current beliefs, shape (num_columns, NUM_OBJECTS)
    - evidence: original sensory evidence (doesn't change)
    - vote_weight: float 0-1, how much to trust neighbors
    Returns:
    - new_beliefs: updated beliefs after voting
    """
    num_columns = beliefs.shape[0]
    new_beliefs = np.zeros_like(beliefs)

    for col in range(num_columns):
        # TODO 1: Calculate this column's own contribution
        own_contribution = (1 - vote_weight) * evidence[col]

        # TODO 2: Get beliefs from OTHER columns (not this one)
        other_indices = [i for i in range(num_columns) if i != col]
        other_beliefs = beliefs[other_indices]

        # TODO 3: Calculate average vote from others
        avg_vote = other_beliefs.mean(axis=0)

        # TODO 4: Calculate vote contribution
        vote_contribution = vote_weight * avg_vote

        # TODO 5: Combine own + vote contributions
        new_beliefs[col] = own_contribution + vote_contribution

    # TODO 6: Normalize rows
    row_sums = new_beliefs.sum(axis=1, keepdims=True)
    new_beliefs = new_beliefs / row_sums

    return new_beliefs

def run_voting(evidence, max_iterations=20, vote_weight=0.3,
               consensus_threshold=0.9):
    """
    Run voting until consensus or max iterations.
    Returns: (history, final_object, final_confidence, iterations_used)
    """

    # TODO 1: Initialize beliefs
    beliefs = initialize_beliefs(evidence)

    # TODO 2: Start history list with initial beliefs
    history = [beliefs.copy()]

    for iteration in range(max_iterations):

        # TODO 3: Do one voting round
        beliefs = voting_round(beliefs, evidence, vote_weight)

        # TODO 4: Add to history
        history.append(beliefs.copy())

        # TODO 5: Check consensus
        predictions = np.argmax(beliefs, axis=1)
        vote_counts = np.bincount(predictions, minlength=NUM_OBJECTS)
        most_common = vote_counts.argmax()
        agreement = (predictions == most_common).mean()

        # TODO 6: Break if consensus reached
        if agreement >= consensus_threshold:
            print(f"Consensus at iteration {iteration + 1}!")
            break

    # Calculate final result
    final_votes = beliefs.mean(axis=0)  # Average across columns
    final_object = final_votes.argmax()
    final_confidence = final_votes[final_object]
    iterations_used = len(history) - 1

    return history, final_object, final_confidence, iterations_used

def visualize_consensus(history, true_object):
    """Visualize how consensus emerges."""

    # Calculate average confidence per object at each iteration
    avg_confidence = np.array([h.mean(axis=0) for h in history])
    # Shape: (num_iterations, NUM_OBJECTS)

    # TODO 1: Calculate agreement at each time step
    agreement_over_time = []
    for h in history:
        predictions = np.argmax(h, axis=1)
        counts = np.bincount(predictions, minlength=NUM_OBJECTS)
        agreement = counts.max() / len(predictions)
        agreement_over_time.append(agreement)

    # Create figure with 2 subplots
    fig, axes = plt.subplots(1, 2, figsize=(12, 4))

    # TODO 2: Plot 1 - Confidence over time
    ax1 = axes[0]
    for obj_idx in range(NUM_OBJECTS):
        # Make true object bold, others dashed
        if obj_idx == true_object:
            style = '-'
            width = 3
        else:
            style = '--'
            width = 1

        ax1.plot(avg_confidence[:, obj_idx], style,
                 linewidth=width, label=OBJECTS[obj_idx])

    ax1.set_xlabel('Voting Round')
    ax1.set_ylabel('Average Confidence')
    ax1.set_title('Confidence Evolution')
    ax1.legend()
    ax1.set_ylim(0, 1)
    ax1.grid(True, alpha=0.3)

    # TODO 3: Plot 2 - Agreement over time
    ax2 = axes[1]
    ax2.plot(agreement_over_time, 'b-', linewidth=2, marker='o')
    ax2.axhline(y=0.9, color='r', linestyle='--', label='Threshold')

    ax2.set_xlabel('Voting Round')
    ax2.set_ylabel('Column Agreement')
    ax2.set_title('Consensus Emergence')
    ax2.legend()
    ax2.set_ylim(0, 1.05)
    ax2.grid(True, alpha=0.3)

    plt.tight_layout()
    plt.savefig('consensus_plot.png', dpi=150)
    plt.show()

    print("Plot saved as consensus_plot.png")

Part 5: Analysis & Reflection¶

5.1 Experiment Analysis¶

Question 1: Effects of Noise (Experiment 1)¶

  1. How did noise affect accuracy and confidence?
  • High noise decreased both accuracy and confidence. Columns disagreed more often, which led to more errors and lower certainty.
  1. Why does noise make consensus harder? (Think about what each column "sees")
  • Noise makes consensus harder because each column "sees" slightly different inputs, so their evidence conflicts.
  1. Did the system still work? What does this suggest about robustness?
  • The system still worked, suggesting it is somewhat robust to noise, though performance declines as noise increases.

Question 2: Number of Columns (Experiment 2)¶

  1. Did more columns improve performance? How?
  • More columns improved performance by averaging out individual errors. Consensus became more reliable with additional columns.
  1. The brain has ~150,000 cortical columns. What advantage might this provide?
  • The brain has ~150,000 cortical columns, which likely provides redundancy and robustness so mistakes in some columns don't ruin overall decisions.
  1. Are there any downsides to having more columns?
  • Downsides of more columns include increased computation and potential conflicts that can slow consensus if not properly balanced.

Question 3: Vote Weight Trade-off (Experiment 3)¶

  1. What is the optimal vote_weight and why?
  • Optimal vote_weight is around 0.5, balancing individual evidence and social influence for stable consensus.
  1. What's wrong with vote_weight = 0.1? (columns don't listen to each other)
  • vote_weight = 0.1 is too low: columns act independently, slowing consensus and reducing accuracy.
  1. What's wrong with vote_weight = 0.9? (columns ignore their own evidence)
  • vote_weight = 0.9 is too high: columns conform too much, ignoring their own evidence and potentially spreading errors.
  1. What general principle does this illustrate about balancing individual vs. collective information?
  • This shows that effective decision-making requires balancing individual knowledge with collective input.

5.2 Connection to Neuroscience¶

Question 4: Distributed Representation¶

  1. How does this relate to the Thousand Brains Theory?
  • This has to do with the Thousand Brains Theory because, in the simulation as well as in the theory, the answer is not known by any one column. Rather, many columns make their own guesses based on incomplete information, and the final answer comes from taking all these different viewpoints together, much as the cortical columns in the brain do.
  1. What would happen if you only had ONE column?
  • With only one column, the system would be very error prone.
  1. Why might the brain use this distributed approach instead of a single"decision center"?
  • The brain uses distributed processing to increase robustness, flexibility, and accuracy instead of relying on a single decision center.

Question 5: Lateral Connections¶

  1. What is the purpose of these connections in your simulation?
  • Lateral connections allow columns to influence each other through voting, simulating communication in the cortex.
  1. What would happen if you set vote_weight = 0 (no connections)?
  • If vote_weight = 0, columns act independently, slowing consensus and reducing reliability.
  1. In the real brain, why might columns need to communicate?
  • In the brain, columns communicate to integrate evidence, reduce uncertainty, and make better decisions.

5.3 Limitations¶

Our Simulation Real Cortical Columns Why It Matters

Simple probability vectors ~100,000 neurons per column Real columns encode complex patterns, not just simplified probabilities.
All columns connected Sparse local connections Real brains mostly have local connections, affecting information flow.
No reference frames Grid cell-based reference frames Spatial and temporal context helps encode object location and orientation.

Pre-defined objects Learns continuously The brain learns from experience; the simulation uses fixed objects.

Question 6: Model Validity¶

  1. What aspects of real cortical processing does this simulation capture well?
  • The simulation captures distributed representation, consensus-building, and balancing individual vs. collective information.
  1. What important aspects does it completely miss?
  • It misses scale, continuous learning, temporal dynamics, and complex connectivity of real cortical columns.
  1. Would you trust this simulation to make predictions about the brain? Why or why not?
  • I would not fully trust it to predict real brain behavior because it simplifies many important aspects, but it is useful for understanding general principles of cortical decision-making.
